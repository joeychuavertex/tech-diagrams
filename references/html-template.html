📋 Template Structure Reference

This template demonstrates the complete interactive HTML shell structure for architecture diagrams. Key components:

1. SVG Container with Defs

<svg id="diagram" viewBox="0 0 1000 500"> <defs> <style id="theme-styles"> /* CSS variables for theming */ :root { --bg: #fafafa; --accent: #2E75B6; } svg { background: var(--bg); } </style> <marker id="arrow"> <polygon fill="var(--accent)" /> </marker> </defs> </svg>

2. Theme Switching JavaScript

function switchTheme(themeName) { const themes = { corporate: { bg: '#fafafa', accent: '#2E75B6' }, blueprint: { bg: '#0f172a', accent: '#60a5fa' } }; const theme = themes[themeName]; const style = document.getElementById('theme-styles'); style.textContent = `:root { --bg: ${theme.bg}; --accent: ${theme.accent}; }`; }

3. SVG Download Function

function downloadSVG() { const svg = document.getElementById('diagram'); const serializer = new XMLSerializer(); const data = serializer.serializeToString(svg); const blob = new Blob([data], { type: 'image/svg+xml' }); const url = URL.createObjectURL(blob); const link = document.createElement('a'); link.href = url; link.download = 'diagram.svg'; link.click(); }
